def _clean_display_text(value):
    """清理 Excel / pandas 常見的空值字串，供地圖顯示使用。"""
    if value is None:
        return ""
    try:
        if pd.isna(value):
            return ""
    except Exception:
        pass
    text = str(value).strip()
    if text.lower() in {"nan", "none", "nat", "null"}:
        return ""
    return text


def _is_land_query_case(row):
    """判斷案件是否為「地號」查詢，供 Popup 是否顯示面積欄位。

    優先使用 Cell 2 已建立的「查詢方式」；若舊版資料沒有該欄位，
    則依 Cell 2 相同規則回推：有地址＝地址案件；否則有地段＋地號＝地號案件。
    """
    if not isinstance(row, (dict, pd.Series)):
        return False

    query_mode = _clean_display_text(row.get("查詢方式", ""))
    if query_mode:
        return query_mode == "地號"

    address = _clean_display_text(row.get("地址", ""))
    section = _clean_display_text(row.get("地段", ""))
    land_no = _clean_display_text(row.get("地號", ""))
    return (not address) and bool(section) and bool(land_no)



def _cell4_case_profile(cell4_results):
    seen = []
    for _mode, _payload in cell4_results.items():
        _wdf = _payload.get("work_df") if isinstance(_payload, dict) else None
        if isinstance(_wdf, pd.DataFrame) and not _wdf.empty:
            seen.append(_wdf.copy())

    if not seen:
        return {"total": 0, "land": 0, "address": 0, "unknown": 0}

    # MOTOR / CAR 可能都是同一批資料，只取第一份即可。
    _df0 = seen[0]
    _land = 0
    _address = 0
    _unknown = 0

    for _, _row in _df0.iterrows():
        _mode = _clean_display_text(_row.get("查詢方式", ""))
        if _mode == "地號":
            _land += 1
        elif _mode == "地址":
            _address += 1
        elif _is_land_query_case(_row):
            _land += 1
        else:
            _unknown += 1

    return {
        "total": len(_df0),
        "land": _land,
        "address": _address,
        "unknown": _unknown,
    }

def google_maps_search_url(lat, lon):
    return (
        "https://www.google.com/maps/search/?api=1&query="
        f"{float(lat):.8f},{float(lon):.8f}"
    )


def google_maps_direction_url(lat, lon, vehicle_mode="CAR"):
    mode = str(vehicle_mode or "CAR").upper()
    travelmode = "two-wheeler" if mode == "MOTOR" else "driving"
    return (
        "https://www.google.com/maps/dir/?api=1"
        f"&destination={float(lat):.8f},{float(lon):.8f}"
        f"&travelmode={travelmode}&dir_action=navigate"
    )

def _format_popup_area(value):
    """將面積整理成適合 Popup 顯示的文字；空值保持空白。"""
    text = _clean_display_text(value)
    if not text:
        return ""
    try:
        number = float(text)
        if math.isfinite(number):
            formatted = f"{number:.6f}".rstrip("0").rstrip(".")
            return formatted or "0"
    except Exception:
        pass
    return text

def _haversine_segment_m(lat1, lon1, lat2, lon2):
    r = 6371008.8
    p1 = math.radians(lat1)
    p2 = math.radians(lat2)
    dp = math.radians(lat2 - lat1)
    dl = math.radians(lon2 - lon1)
    a = (
        math.sin(dp / 2) ** 2
        + math.cos(p1) * math.cos(p2) * math.sin(dl / 2) ** 2
    )
    return r * 2 * math.asin(math.sqrt(min(1.0, max(0.0, a))))


def _bearing_deg(lat1, lon1, lat2, lon2):
    p1 = math.radians(lat1)
    p2 = math.radians(lat2)
    dl = math.radians(lon2 - lon1)
    y = math.sin(dl) * math.cos(p2)
    x = (
        math.cos(p1) * math.sin(p2)
        - math.sin(p1) * math.cos(p2) * math.cos(dl)
    )
    return (math.degrees(math.atan2(y, x)) + 360.0) % 360.0


def _interpolate_on_polyline(locations, target_m):
    if not locations or len(locations) < 2:
        return None
    remaining = float(target_m)
    for i in range(len(locations) - 1):
        lat1, lon1 = map(float, locations[i])
        lat2, lon2 = map(float, locations[i + 1])
        seg_m = _haversine_segment_m(lat1, lon1, lat2, lon2)
        if seg_m <= 0:
            continue
        if remaining <= seg_m:
            t = max(0.0, min(1.0, remaining / seg_m))
            lat = lat1 + (lat2 - lat1) * t
            lon = lon1 + (lon2 - lon1) * t
            return lat, lon, _bearing_deg(lat1, lon1, lat2, lon2)
        remaining -= seg_m
    lat1, lon1 = map(float, locations[-2])
    lat2, lon2 = map(float, locations[-1])
    return lat2, lon2, _bearing_deg(lat1, lon1, lat2, lon2)


def _polyline_length_m(locations):
    if not locations or len(locations) < 2:
        return 0.0
    return sum(
        _haversine_segment_m(float(a[0]), float(a[1]), float(b[0]), float(b[1]))
        for a, b in zip(locations[:-1], locations[1:])
    )


def _destination_point(lat, lon, bearing_deg, distance_m):
    R = 6371008.8
    br = math.radians(bearing_deg)
    lat1 = math.radians(lat)
    lon1 = math.radians(lon)
    ang = float(distance_m) / R
    lat2 = math.asin(
        math.sin(lat1) * math.cos(ang)
        + math.cos(lat1) * math.sin(ang) * math.cos(br)
    )
    lon2 = lon1 + math.atan2(
        math.sin(br) * math.sin(ang) * math.cos(lat1),
        math.cos(ang) - math.sin(lat1) * math.sin(lat2),
    )
    return math.degrees(lat2), ((math.degrees(lon2) + 540) % 360) - 180


def _make_arrow_triangle(lat, lon, bearing_deg, length_m=90.0, width_m=32.0):
    tip = _destination_point(lat, lon, bearing_deg, length_m / 2.0)
    tail = _destination_point(
        lat, lon, (bearing_deg + 180.0) % 360.0, length_m / 2.0
    )
    left = _destination_point(
        tail[0], tail[1], (bearing_deg + 90.0) % 360.0, width_m / 2.0
    )
    right = _destination_point(
        tail[0], tail[1], (bearing_deg - 90.0) % 360.0, width_m / 2.0
    )
    return [[tip[0], tip[1]], [left[0], left[1]], [right[0], right[1]]]


def add_route_line_with_arrows(
    layer,
    locations,
    color,
    weight=6,
    opacity=0.9,
    tooltip=None,
    dash_array=None,
    arrow_spacing_m=750.0,
    arrow_layer=None,
):
    if not locations or len(locations) < 2:
        return

    folium.PolyLine(
        locations=locations,
        color=color,
        weight=weight,
        opacity=opacity,
        dash_array=dash_array,
        tooltip=tooltip,
    ).add_to(layer)

    route_m = _polyline_length_m(locations)
    if route_m <= 100:
        return

    spacing = max(300.0, float(arrow_spacing_m))
    arrow_count = min(int(route_m // spacing), 40)
    if arrow_count <= 0:
        targets = [route_m * 0.5]
    else:
        actual_spacing = route_m / (arrow_count + 1)
        targets = [actual_spacing * (i + 1) for i in range(arrow_count)]

    arrow_target = arrow_layer if arrow_layer is not None else layer

    for target_m in targets:
        result = _interpolate_on_polyline(locations, target_m)
        if result is None:
            continue
        lat, lon, bearing = result
        folium.Polygon(
            locations=_make_arrow_triangle(lat, lon, bearing),
            color=color,
            weight=1,
            opacity=0.95,
            fill=True,
            fill_color=color,
            fill_opacity=0.95,
            interactive=False,
        ).add_to(arrow_target)


def _prepare_cadastral_map_gdf(cad_gdf, route_order_df):
    gdf = cad_gdf.copy()

    if "點索引" not in gdf.columns:
        if "列號" in gdf.columns:
            gdf["點索引"] = pd.to_numeric(gdf["列號"], errors="coerce") - 1
        else:
            gdf["點索引"] = np.arange(len(gdf), dtype=int)

    gdf["點索引"] = pd.to_numeric(gdf["點索引"], errors="coerce")

    route = route_order_df.copy()
    if "點索引" not in route.columns:
        if "原始_df_index" in route.columns:
            route["點索引"] = pd.to_numeric(
                route["原始_df_index"], errors="coerce"
            )
        else:
            raise RuntimeError(
                "Cell 4 route_order_df 缺少「點索引」／「原始_df_index」。"
            )

    route["點索引"] = pd.to_numeric(route["點索引"], errors="coerce")

    # ------------------------------------------------------------
    # Cell 4 加入「勘查狀態」篩選後，路線 work_df 的點索引會重新編排。
    # Cell 3 的「點索引」仍對應原始 Excel 列序，因此不能直接拿兩者
    # 對接。優先使用：
    #   Cell 4 route_order_df「原始_df_index」
    #   對 Cell 3「列號 - 1」
    # 讓已完成案件被排除後，Cell 5 仍能把正確 Polygon 對回正確案件。
    # ------------------------------------------------------------
    route_meta_cols = [
        c for c in [
            "點索引", "原始_df_index",
            "工作日", "日內序號", "總路線序號", "角色"
        ]
        if c in route.columns
    ]
    route_meta = route[route_meta_cols].copy()

    if "原始_df_index" in route_meta.columns and "列號" in gdf.columns:
        route_meta["_cad_match_idx"] = pd.to_numeric(
            route_meta["原始_df_index"], errors="coerce"
        )
        gdf["_cad_match_idx"] = pd.to_numeric(
            gdf["列號"], errors="coerce"
        ) - 1

        route_meta = route_meta.drop_duplicates("_cad_match_idx")

        gdf = gdf.merge(
            route_meta.drop(columns=["原始_df_index"], errors="ignore"),
            on="_cad_match_idx",
            how="left",
            suffixes=("", "_route"),
        )
        gdf = gdf.drop(columns=["_cad_match_idx"], errors="ignore")
    else:
        route_meta = route_meta.drop_duplicates("點索引")
        gdf = gdf.merge(
            route_meta,
            on="點索引",
            how="left",
            suffixes=("", "_route"),
        )

    preferred = [
        "案件ID", "點索引", "縣市", "鄉鎮市區", "地段", "地號", "查詢地號", "名稱", "勘查狀態", "固定起點",
        "面積_m2", "解析度分類", "定位點位於宗地", "中心偏移_m",
        "幾何狀態", "GeometryType", "工作日", "日內序號",
        "總路線序號", "角色", "geometry",
    ]
    cols = [c for c in preferred if c in gdf.columns]
    gdf = gdf[cols].copy()

    if "工作日" not in gdf.columns:
        gdf["工作日"] = np.nan
    gdf["工作日"] = pd.to_numeric(gdf["工作日"], errors="coerce")

    gdf = gdf[gdf.geometry.notna()].copy()
    gdf = gdf[
        gdf.geometry.geom_type.isin(["Polygon", "MultiPolygon"])
    ].copy()

    # 已完成且非固定起點：Cell 5 不顯示其地籍 Polygon。
    # 固定起點即使已完成，仍保留。
    if "勘查狀態" in gdf.columns:
        completed_mask = gdf["勘查狀態"].astype(str).str.strip().eq("已完成")
        fixed_mask = pd.Series(False, index=gdf.index)
        if "固定起點" in gdf.columns:
            fixed_mask = gdf["固定起點"].astype(str).str.strip().eq("是")
        if "角色" in gdf.columns:
            fixed_mask = fixed_mask | gdf["角色"].astype(str).str.strip().eq("固定起點")
        gdf = gdf[(~completed_mask) | fixed_mask].copy()

    return gdf


def _cadastral_geojson(gdf):
    serial = gdf.copy()
    for col in serial.columns:
        if col == "geometry":
            continue
        if pd.api.types.is_numeric_dtype(serial[col]):
            serial[col] = serial[col].where(pd.notna(serial[col]), None)
        else:
            serial[col] = serial[col].map(
                lambda v: "" if pd.isna(v) else str(v)
            )
    return json.loads(serial.to_json())



def _norm_search_text(value):
    """搜尋用正規化：轉小寫並移除空白、連字號及常見分隔符。"""
    if value is None:
        return ""
    s = str(value).strip().lower()
    for ch in [" ", "　", "-", "－", "_", "/", "\\", "."]:
        s = s.replace(ch, "")
    return s


def _safe_num(value, default=0.0):
    try:
        if pd.isna(value):
            return default
        return float(value)
    except Exception:
        return default


def _standardized_cadastral_name(row):
    """建立外勤地圖／勘查點資料使用的正式顯示名稱。

    【v32.8.5 名稱規則】
    1. 地號查詢案件：固定組合成「縣市＋鄉鎮市區＋地段＋8碼地號」。
       例：桃園市八德區茄苳段07360000
       不使用 Cell 4 舊版可能產生的「桃園市-茄苳-7360000.0」名稱。
    2. 地址查詢案件：維持原本案件名稱優先；沒有名稱時再組合完整地址。
    3. 優先使用 Cell 2 → Cell 3 → Cell 4 共用的「顯示名稱」；無該欄位時才使用本函式備援組合。
    """
    if row is None:
        return ""

    def norm_text(value):
        return _clean_display_text(value)

    def compact_text(value):
        return re.sub(r"[\s\u3000\-－_/\\\\,，。．:：;；]", "", norm_text(value))

    def normalize_land_no_display(value):
        text = norm_text(value)
        if not text:
            return ""

        # pandas / Excel 常將整數地號讀成 7360000.0；先去除尾端 .0。
        if re.fullmatch(r"\d+\.0+", text):
            text = text.split(".", 1)[0]

        # 純數字地號統一為 8 碼，例如 7360000 → 07360000。
        if re.fullmatch(r"\d+", text):
            return text.zfill(8)

        # 已有常見分隔格式時，不任意改變其語意；僅去除空白。
        return re.sub(r"[\s\u3000]", "", text)

    def normalize_section_display(value):
        text = norm_text(value)
        if not text:
            return ""
        # 只移除前後空白；不改變 NLSC 已提供的正式段名。
        return text

    city = norm_text(row.get("縣市", ""))
    area = norm_text(row.get("鄉鎮市區", ""))
    section = normalize_section_display(row.get("地段", ""))
    query_mode = norm_text(row.get("查詢方式", ""))

    if not query_mode:
        address_probe = norm_text(row.get("地址", ""))
        land_probe = norm_text(row.get("地號", ""))
        query_mode = "地址" if address_probe else ("地號" if section and land_probe else "")

    # v32.8.5：上游已有統一「顯示名稱」時直接使用。
    # 地址案件、地號案件均以此作為 Cell 5 的第一來源。
    upstream_display_name = norm_text(row.get("顯示名稱", ""))
    if upstream_display_name:
        return upstream_display_name

    # ========================================================
    # ① 地號案件：正式地籍名稱備援組合
    # ========================================================
    # 不再採用 Cell 4 舊版「名稱」欄位的連字號／浮點數格式。
    if query_mode == "地號":
        land_candidates = [
            "標準化後之地號(全部)",
            "標準化後地號(全部)",
            "標準化後之地號",
            "標準化後地號",
            "完整標準化地號",
            "完整地號",
            "完整查詢地號",
            "查詢地號",
            "地號",
        ]
        land_no = ""
        for field in land_candidates:
            value = normalize_land_no_display(row.get(field, ""))
            if value:
                land_no = value
                break

        # 優先組合：縣市＋鄉鎮市區＋地段＋地號。
        if city and area and section and land_no:
            return f"{city}{area}{section}{land_no}"

        # 若前述欄位有缺漏，再使用可辨識的完整地號欄位。
        for field in land_candidates:
            value = norm_text(row.get(field, ""))
            if value and compact_text(value) not in {
                compact_text(city),
                compact_text(area),
                compact_text(section),
            }:
                return value

        combined = "".join(v for v in (city, area, section, land_no) if v)
        if combined:
            return combined

    # ========================================================
    # ② 地址案件：保留原本案件名稱／完整地址邏輯
    # ========================================================
    address = norm_text(row.get("地址", ""))
    name_candidates = [
        "名稱",
        "校正後名稱",
        "顯示名稱",
        "點位名稱",
        "完整名稱",
        "標準化名稱",
        "校正後地籍名稱",
        "完整地籍名稱",
        "標準化地籍名稱",
        "校正名稱",
    ]

    generic_values = {
        compact_text(city),
        compact_text(area),
        compact_text(f"{city}{area}"),
        compact_text(section),
        compact_text(f"{city}{area}{section}"),
        "固定起點",
        "起點",
        "nan",
        "none",
        "null",
    }

    for field in name_candidates:
        value = norm_text(row.get(field, ""))
        if not value:
            continue
        compact = compact_text(value)
        if compact in generic_values:
            continue
        return value

    address_candidates = [
        "標準化後之地址(全部)",
        "標準化後地址(全部)",
        "標準化後之地址",
        "標準化後地址",
        "完整標準化地址",
        "完整地址",
        "完整查詢地址",
        "查詢地址",
        "命中地址",
        "地址",
        "校正後地址",
        "位置",
        "地點",
    ]

    for field in address_candidates:
        value = norm_text(row.get(field, ""))
        if not value:
            continue
        compact = compact_text(value)
        if compact in generic_values:
            continue
        combined_address = "".join(v for v in (city, area, value) if v).strip()
        return combined_address or value

    # ========================================================
    # ③ 最後 fallback
    # ========================================================
    land_no = normalize_land_no_display(row.get("地號", ""))
    combined = "".join(v for v in (city, area, section, land_no) if v).strip()
    if combined and compact_text(combined) not in generic_values:
        return combined

    return city or area or section or land_no or address

def _build_work_lookup(work_df):
    """建立 Cell 4 工作資料的 canonical lookup。

    route_order_df 主要存放路線順序；案件「名稱」等原始欄位以 work_df 為準。

    v33.19 地址案件修正：
    Cell 4 route_order_df 同時可能提供「原始_df_index」與「點索引」。
    「原始_df_index」才是回到 work_df 原始列序的穩定鍵，因此另外建立
    lookup_orig，供 _canonical_route_row 優先使用。
    """
    lookup_idx = {}
    lookup_case = {}
    lookup_orig = {}
    if not isinstance(work_df, pd.DataFrame) or work_df.empty:
        return lookup_idx, lookup_case, lookup_orig
    for work_pos, wr in work_df.iterrows():
        pidx = _safe_num(wr.get("點索引", -1), -1)
        if pidx >= 0:
            lookup_idx[int(pidx)] = wr

        # Cell 4 正式 work_df 使用「_原始_df_index」保存 Cell 2 原始列索引。
        # 舊版資料可能使用「原始_df_index」，兩者都支援；
        # 若兩者皆不存在，最後才退回 work_df 的 DataFrame index。
        orig_idx = _safe_num(
            wr.get("_原始_df_index", wr.get("原始_df_index", work_pos)),
            -1,
        )
        if orig_idx >= 0:
            lookup_orig[int(orig_idx)] = wr

        case_id = _clean_display_text(wr.get("案件ID", ""))
        if case_id:
            lookup_case[case_id] = wr
    return lookup_idx, lookup_case, lookup_orig


def _canonical_route_row(
    row,
    work_lookup_idx=None,
    work_lookup_case=None,
    work_lookup_orig=None,
):
    """把 route_order_df 單列補回 work_df 的 canonical 案件資料。

    回補優先順序：
    1. route_order_df 的「原始_df_index」
    2. route_order_df 的「點索引」
    3. route_order_df 的「案件ID」

    其中第 1 項可避免案件經 Cell 4 篩選／重排後，「點索引」不再
    與 work_df 原始列序一致，特別是地址案件的案件屬性因此遺失。
    """
    if isinstance(row, pd.Series):
        data = row.to_dict()
    elif isinstance(row, dict):
        data = dict(row)
    else:
        try:
            data = dict(row)
        except Exception:
            return row

    work_lookup_idx = work_lookup_idx or {}
    work_lookup_case = work_lookup_case or {}
    work_lookup_orig = work_lookup_orig or {}

    src = None

    # ① 最優先：Cell 4 route_order_df 的原始 Excel 列索引。
    orig_idx = _safe_num(data.get("原始_df_index", -1), -1)
    if orig_idx >= 0:
        src = work_lookup_orig.get(int(orig_idx))

    # ② 其次才使用「點索引」。
    if src is None:
        pidx = _safe_num(data.get("點索引", -1), -1)
        if pidx >= 0:
            src = work_lookup_idx.get(int(pidx))

    # ③ 最後使用穩定案件ID。
    if src is None:
        case_id = _clean_display_text(data.get("案件ID", ""))
        if case_id:
            src = work_lookup_case.get(case_id)

    if src is not None:
        # 案件原始欄位一定以 work_df 為準；路線欄位則保留 route_order_df。
        for field in (
            "名稱", "勘查狀態", "縣市", "鄉鎮市區", "地段", "地號", "查詢地號", "地址",
            "校正後名稱", "顯示名稱", "點位名稱", "完整名稱",
            "標準化名稱", "校正名稱", "校正後地籍名稱",
            "完整地籍名稱", "標準化地籍名稱",
            "標準化後之地號(全部)", "標準化後地號(全部)",
            "標準化後之地號", "標準化後地號",
            "完整標準化地號", "完整地號", "完整查詢地號",
            "校正後地址", "查詢地址", "位置", "地點",
            "標準化後之地址(全部)", "標準化後地址(全部)",
            "標準化後之地址", "標準化後地址",
            "完整標準化地址", "完整地址",
            "定位來源", "完整查詢地址", "命中地址", "案件ID",
            "查詢方式", "本筆面積(公頃)", "申請面積(公頃)",
        ):
            value = _clean_display_text(src.get(field, ""))
            if value:
                data[field] = value
    return data


def _resolve_start_display_name(route_order_df, fallback_name, work_df=None, cadastral_map_gdf=None):
    """固定起點顯示名稱；Cell 4 payload 的 start_name 優先。"""
    # ① Cell 4 start_name 是真正的固定起點名稱，絕對優先。
    fallback = _clean_display_text(fallback_name)
    if fallback and fallback not in {"固定起點", "起點"}:
        return fallback

    # ② 其次找 work_df 中的固定起點
    if isinstance(work_df, pd.DataFrame) and not work_df.empty:
        candidates = work_df.copy()
        if "角色" in candidates.columns:
            fixed = candidates[candidates["角色"].astype(str).str.strip().eq("固定起點")]
            if not fixed.empty:
                candidates = fixed
        if "點索引" in candidates.columns:
            fixed_idx = candidates[pd.to_numeric(candidates["點索引"], errors="coerce").eq(0)]
            if not fixed_idx.empty:
                candidates = fixed_idx
        for _, row in candidates.iterrows():
            name = _standardized_cadastral_name(row)
            if name and name not in {"固定起點", "起點"}:
                return name

    # ③ route_order_df 的固定起點
    if isinstance(route_order_df, pd.DataFrame) and not route_order_df.empty:
        candidates = route_order_df.copy()
        if "角色" in candidates.columns:
            fixed = candidates[candidates["角色"].astype(str).str.strip().eq("固定起點")]
            if not fixed.empty:
                candidates = fixed
        if "點索引" in candidates.columns:
            fixed_idx = candidates[pd.to_numeric(candidates["點索引"], errors="coerce").eq(0)]
            if not fixed_idx.empty:
                candidates = fixed_idx
        for _, row in candidates.iterrows():
            name = _standardized_cadastral_name(row)
            if name and name not in {"固定起點", "起點"}:
                return name

    # ④ 最後找地址
    for df in (cadastral_map_gdf, route_order_df):
        if isinstance(df, pd.DataFrame) and not df.empty:
            for _, row in df.iterrows():
                for field in ("地址", "校正後地址", "查詢地址", "位置", "地點"):
                    value = _clean_display_text(row.get(field, ""))
                    if value:
                        return value
    return fallback or "固定起點"

def _build_summary(payload, start_name):
    actual_batches = payload.get("actual_batches", []) or []
    route_order_df = payload.get("route_order_df")
    check_count = 0
    if isinstance(route_order_df, pd.DataFrame) and not route_order_df.empty:
        if "點索引" in route_order_df.columns:
            point_series = pd.to_numeric(
                route_order_df["點索引"], errors="coerce"
            ).dropna()
            point_series = point_series[point_series >= 0]
            raw_count = int(point_series.nunique())
            # 固定起點永遠不算勘查土地數量；實際查核點數 = route_order_df 中非起點的唯一點索引數。
            check_count = max(0, raw_count - 1)
        else:
            check_count = max(0, int(len(route_order_df)) - 1)
    work_days = len(actual_batches)
    closed_km = 0.0
    for info in actual_batches:
        closed_km += _safe_num(info.get("distance", 0)) / 1000.0
        closed_km += _safe_num(info.get("return_distance", 0)) / 1000.0
    one_way_km = _safe_num(payload.get("all_in_one_actual_distance", 0)) / 1000.0
    one_way_hr = _safe_num(payload.get("all_in_one_actual_drive_min", 0)) / 60.0
    return {
        "交通工具": str(payload.get("VEHICLE_LABEL", payload.get("VEHICLE_MODE", ""))),
        "查核點": check_count,
        "工作日": work_days,
        "每日閉合路程_km": closed_km,
        "一次完成路程_km": one_way_km,
        "一次完成駕駛_hr": one_way_hr,
        "固定起點": str(start_name),
    }


def _build_static_layer_panel_html(layer_items):
    """在 HTML 產生階段直接建立「地圖顯示內容」清單。

    這是 JS 初始化失敗時的安全保底：即使瀏覽器尚未執行 Cell 5 的
    自訂面板 JavaScript，使用者仍然可以看到所有可用圖層。
    """
    items = list(layer_items or [])
    bases = [x for x in items if x.get("type") == "base"]
    daily = [x for x in items if x.get("group") == "daily"]
    all_items = [x for x in items if x.get("group") == "all"]
    number_items = [x for x in items if x.get("group") == "number"]
    other = [
        x for x in items
        if x.get("type") == "overlay"
        and x.get("group") not in {"daily", "all", "number"}
    ]

    def row(item, idx, prefix):
        item_type = str(item.get("type") or "overlay")
        input_type = "radio" if item_type == "base" else "checkbox"
        checked = " checked" if bool(item.get("show")) else ""
        name_attr = ' name="cell5-base-map-static"' if item_type == "base" else ""
        var_name = escape(str(item.get("var") or ""), quote=True)
        label = escape(str(item.get("label") or ""))
        item_group = escape(str(item.get("group") or ""), quote=True)
        return (
            '<div class="cell5-layer-row"'
            f' data-layer-var="{var_name}" data-layer-type="{item_type}"'
            f' data-layer-group="{item_group}">'
            f'<input id="cell5-layer-static-{prefix}-{idx}" type="{input_type}"{name_attr}{checked} />'
            f'<label for="cell5-layer-static-{prefix}-{idx}">{label}</label>'
            '</div>'
        )

    html = []
    html.append('<div class="cell5-layer-section-title">底圖</div>')
    for i, item in enumerate(bases):
        html.append(row(item, i, "b"))

    html.append('<div class="cell5-layer-section-title" style="margin-top:8px;">路線與勘查順序</div>')
    html.append('<div class="cell5-layer-mode-note">分日、一次完成、勘查編號三者互斥；分日可同時開啟各 DAY 路線。</div>')
    for i, item in enumerate(daily):
        html.append(row(item, i, "d"))
    for i, item in enumerate(all_items):
        html.append(row(item, i, "a"))
    for i, item in enumerate(number_items):
        html.append(row(item, i, "n"))

    html.append('<div class="cell5-layer-section-title" style="margin-top:8px;">其他</div>')
    for i, item in enumerate(other):
        html.append(row(item, i, "o"))

    if not items:
        html.append('<div style="padding:8px;color:#666;font-size:11px;">目前沒有可用的地圖圖層。</div>')
    return "".join(html)

      function isNumberModeActive() {
        try {
          const item = (layerItems || []).find(function(x) {
            return String(x && x.group || '') === 'number';
          });
          if (!item) return false;
          const layer = item.var ? window[item.var] : null;
          return !!(layer && mapObj.hasLayer(layer));

      function getVisibleStartTargets() {
        // 勘查編號模式只有全域編號；固定起點 0 不提供下一點選擇。
        if (isNumberModeActive()) return [];

        const out = [];
        (dailyDayTargets || []).forEach(function(t) {
          const layer = t.route_var ? window[t.route_var] : null;
          if (layer && mapObj.hasLayer(layer) && t.first_marker_var) out.push(t);
        });
        if (!out.length && allInOneStartTarget && allInOneStartTarget.route_var) {
          const layer = window[allInOneStartTarget.route_var];
          if (layer && mapObj.hasLayer(layer) && allInOneStartTarget.first_marker_var) out.push(allInOneStartTarget);
        }
        // 固定起點資訊本身不應因路線圖層暫時未勾選而消失；
        // 若目前沒有可見路線，仍提供所有可用路線的入口。
        if (!out.length) {
          (dailyDayTargets || []).forEach(function(t) {
            if (t && t.first_marker_var) out.push(t);
          });
          if (!out.length && allInOneStartTarget && allInOneStartTarget.first_marker_var) {
            out.push(allInOneStartTarget);
          }
        }
        return out;
      }

      function updateCell5LocationVisual(pos, recenter) {
        try {
          const lat = Number(pos.coords.latitude);
          const lon = Number(pos.coords.longitude);
          const accuracy = Math.max(0, Number(pos.coords.accuracy || 0));
          if (!Number.isFinite(lat) || !Number.isFinite(lon)) return;

          cell5LocationState.last = {lat: lat, lon: lon, accuracy: accuracy};

          if (!cell5LocationState.marker) {
            cell5LocationState.marker = L.marker([lat, lon], {
              icon: L.divIcon({
                className: '',
                html: '<div class="cell5-user-location-marker"></div>',
                iconSize: [18,18],
                iconAnchor: [9,9]
              }),
              zIndexOffset: 60000000,
              interactive: false,
              keyboard: false
            }).addTo(mapObj);
          } else {
            cell5LocationState.marker.setLatLng([lat, lon]);
          }

          if (!cell5LocationState.accuracy) {
            cell5LocationState.accuracy = L.circle([lat, lon], {
              radius: accuracy,
              color: '#1a73e8',
              weight: 1,
              opacity: .32,
              fillColor: '#1a73e8',
              fillOpacity: .07,
              interactive: false
            }).addTo(mapObj);
          } else {
            cell5LocationState.accuracy.setLatLng([lat, lon]);
            cell5LocationState.accuracy.setRadius(accuracy);
          }

          if (recenter || cell5LocationState.firstFix) {
            mapObj.setView([lat, lon], Math.max(mapObj.getZoom(), 16), {animate:true});
            cell5LocationState.firstFix = false;
          }
        } catch(e) {
          try { console.warn('Cell5 目前位置更新失敗', e); } catch(_) {}
        }
      }

      function cell5LocationErrorMessage(err) {
        if (!err) return '目前位置取得失敗';
        if (err.code === 1) return '未允許位置權限';
        if (err.code === 2) return '目前無法取得位置';
        if (err.code === 3) return '定位逾時';
        return String(err.message || '目前位置取得失敗');
      }

      window.cell5StartLocation = function(recenter) {
        try {
          if (!navigator.geolocation) {
            console.warn('此瀏覽器不支援裝置定位');
            return false;
          }

          if (cell5LocationState.watchId !== null) {
            if (cell5LocationState.last && recenter) window.cell5CenterLocation();
            return false;
          }

          const opts = {
            enableHighAccuracy: true,
            maximumAge: 5000,
            timeout: 15000
          };

          cell5LocationState.watchId = navigator.geolocation.watchPosition(
            function(pos) {
              updateCell5LocationVisual(pos, !!recenter);
              recenter = false;
            },
            function(err) {
              console.warn('Cell5 目前位置：' + cell5LocationErrorMessage(err));
            },
            opts
          );

          navigator.geolocation.getCurrentPosition(
            function(pos) {
              updateCell5LocationVisual(pos, !!recenter);
              recenter = false;
            },
            function(err) {
              console.warn('Cell5 目前位置：' + cell5LocationErrorMessage(err));
            },
            opts
          );
        } catch(e) {
          console.warn('Cell5 目前位置啟動失敗', e);
        }
        return false;
      };

        return false;
      };

      function showInspectionPanel(markerVar) {
        const ui = getInspectionPanel();
        if (!ui.panel || !ui.body) return;
        const marker = resolveMarker(markerVar);
        const popup = marker && marker.getPopup ? marker.getPopup() : null;
        const popupContent = popup ? popup.getContent() : '';
        const meta=inspectionMeta[markerVar]||{};
        let content = inspectionRecords[markerVar] || popupContent || '';
        if (content && typeof content === 'object') {
          if (content.outerHTML) content = content.outerHTML;
          else if (content.innerHTML) content = content.innerHTML;
          else if (Array.isArray(content) && content[0]) content = content[0].outerHTML || content[0].innerHTML || '';
        }
        if (meta.display_name) {
          const safeName = '<div class="cell5-inspection-name">' + escapeHtmlText(meta.display_name) + '</div>';
          if (content) content = String(content).replace(/<div class="cell5-inspection-name">[\s\S]*?<\/div>/i, safeName);
          else content = safeName;
        }
        const pointLabel=meta.label||'';
        ui.title.textContent=pointLabel||'勘查點資料';
        ui.body.innerHTML=(content?String(content):'<div>此點沒有可顯示資料。</div>');
        try {
          ui.body.querySelectorAll('.cell5-return-start').forEach(function(el){ el.remove(); });
          if (mapObj && mapObj.closePopup) mapObj.closePopup();
        } catch(e) {}
        ui.panel.classList.add('visible');
        ui.panel.setAttribute('aria-hidden','false');
        inspectionPanelState={kind:'point',markerVar:markerVar};
      }

      function findCadByPointIndex(pointIndex) {
        let found = null;
        if (!cadLayer || typeof cadLayer.eachLayer !== 'function') return null;
        cadLayer.eachLayer(function(layer) {
          if (found) return;
          const p = (layer.feature && layer.feature.properties) || {};
          if (String(p['點索引'] ?? '') === String(pointIndex ?? '')) {
            found = layer;
          }
        });
        return found;
      }

      function highlightCad(pointIndex, caseId) {
        resetActiveCad();
        let layer = findCadByPointIndex(pointIndex);
        if (!layer && caseId) layer = findCadByCaseId(caseId);
        if (!layer) return false;

        activeCad = layer;
        try {
          layer.setStyle({weight: 5, color: '#111111', fillOpacity: 0.28});
          if (layer.bringToFront) layer.bringToFront();
          const b = layer.getBounds();
          if (b && b.isValid()) mapObj.fitBounds(b.pad(0.15));
        } catch(e) {}
        return true;
      }

      function buildCustomLayerPanel() {
        const body = document.getElementById('cell5-custom-layer-body');
        if (!body) return;
        let html = '';
        const bases = (layerItems || []).filter(function(x){ return x.type === 'base'; });
        const daily = (layerItems || []).filter(function(x){ return x.group === 'daily'; });
        const all = (layerItems || []).filter(function(x){ return x.group === 'all'; });
        const number = (layerItems || []).filter(function(x){ return x.group === 'number'; });
        const other = (layerItems || []).filter(function(x){ return x.type === 'overlay' && x.group !== 'daily' && x.group !== 'all' && x.group !== 'number'; });

        html += '<div class="cell5-layer-section-title">底圖</div>';
        bases.forEach(function(item, idx){ html += customLayerRowHtml(item, 'b-' + idx); });
        html += '<div class="cell5-layer-section-title" style="margin-top:8px;">路線與勘查順序</div>';
        html += '<div class="cell5-layer-mode-note">分日、一次完成、勘查編號三者互斥；DAY01/02/03 可以同時開啟。</div>';
        daily.forEach(function(item, idx){ html += customLayerRowHtml(item, 'd-' + idx); });
        all.forEach(function(item, idx){ html += customLayerRowHtml(item, 'a-' + idx); });
        number.forEach(function(item, idx){ html += customLayerRowHtml(item, 'n-' + idx); });
        html += '<div class="cell5-layer-section-title" style="margin-top:8px;">其他</div>';
        other.forEach(function(item, idx){ html += customLayerRowHtml(item, 'o-' + idx); });
        body.innerHTML = html;

        body.querySelectorAll('input').forEach(function(input){
          input.addEventListener('change', function(e){
            const row = input.closest('.cell5-layer-row');
            if (!row) return;
            const varName = row.dataset.layerVar || '';
            const type = row.dataset.layerType || 'overlay';
            const layer = getLayerByVar(varName);
            if (!layer) return;

            if (type === 'base') {
              if (input.checked) {
                bases.forEach(function(baseItem){
                  const b = getLayerByVar(baseItem.var);
                  if (b && b !== layer && mapObj.hasLayer(b)) mapObj.removeLayer(b);
                });
                if (!mapObj.hasLayer(layer)) mapObj.addLayer(layer);
              }
            } else {
              if (input.checked) {
                const item = (layerItems || []).find(function(x){ return x.var === varName; });
                if (item && item.group === 'all') {
                  daily.forEach(function(d){ const dl=getLayerByVar(d.var); if(dl && mapObj.hasLayer(dl)) mapObj.removeLayer(dl); });
                  number.forEach(function(n){ const nl=getLayerByVar(n.var); if(nl && mapObj.hasLayer(nl)) mapObj.removeLayer(nl); });
                } else if (item && item.group === 'daily') {
                  all.forEach(function(a){ const al=getLayerByVar(a.var); if(al && mapObj.hasLayer(al)) mapObj.removeLayer(al); });
                  number.forEach(function(n){ const nl=getLayerByVar(n.var); if(nl && mapObj.hasLayer(nl)) mapObj.removeLayer(nl); });
                } else if (item && item.group === 'number') {
                  daily.forEach(function(d){ const dl=getLayerByVar(d.var); if(dl && mapObj.hasLayer(dl)) mapObj.removeLayer(dl); });
                  all.forEach(function(a){ const al=getLayerByVar(a.var); if(al && mapObj.hasLayer(al)) mapObj.removeLayer(al); });
                }
                mapObj.addLayer(layer);
              } else {
                mapObj.removeLayer(layer);
              }
            }
            syncCustomLayerCheckboxes();
            syncArrowVisibility();
            updateSharedStartPanel();
          });
        });
      }

def _download_text(url, cache_path):
    """下載並快取文字資源；用於產生可單檔開啟的地圖 HTML。"""
    try:
        if os.path.exists(cache_path):
            with open(cache_path, "r", encoding="utf-8") as _f:
                return _f.read()
        with urllib.request.urlopen(url, timeout=30) as _r:
            data = _r.read().decode("utf-8")
        os.makedirs(os.path.dirname(cache_path), exist_ok=True)
        with open(cache_path, "w", encoding="utf-8") as _f:
            _f.write(data)
        return data
    except Exception as _e:
        print(f"[警告] 無法下載外部資源 {url}：{_e}")
        return None

def _make_http_mobile_html(html):
    """確保透過 HTTP 網頁入口回傳的內容一定是可渲染的 HTML。

    這裡不依賴 URL 副檔名，也不設定 Content-Disposition；
    避免 iPhone / Microsoft Edge 把內容判斷成「下載檔案」。
    """
    if not isinstance(html, str):
        html = str(html)

    # 手機 viewport
    new_viewport = (
        '<meta name="viewport" content="width=device-width, '
        'initial-scale=1.0, viewport-fit=cover, maximum-scale=5.0" />'
    )
    if re.search(r'<meta\s+name=["\']viewport["\'][^>]*>', html, flags=re.I):
        html = re.sub(
            r'<meta\s+name=["\']viewport["\'][^>]*>',
            lambda _m: new_viewport,
            html,
            count=1,
            flags=re.I,
        )
    elif "</head>" in html.lower():
        html = html.replace("</head>", new_viewport + "</head>", 1)

    # 確保 iOS web-app metadata 存在
    if "apple-mobile-web-app-capable" not in html:
        html = html.replace(
            "</head>",
            (
                '<meta name="apple-mobile-web-app-capable" content="yes" />'
                '<meta name="apple-mobile-web-app-status-bar-style" content="default" />'
                '<meta name="mobile-web-app-capable" content="yes" />'
                "</head>"
            ),
            1,
        )

    # 防止頁面被瀏覽器視為檔案下載／附件。
    # 這裡只處理 HTML 內容，不碰 Leaflet 程式碼本身。
    return html

def _fetch_nlsc_emap_wms_snapshot(bbox, target_width=3600):
    """取得 NLSC EMAP 高解析度 WMS 拼接快照。

    手機直接開啟 HTML 時，file:// 模式無法穩定依賴即時圖磚，
    因此改成多張較高解析度 WMS 圖片拼接成一張底圖後內嵌到 HTML。

    官方 NLSC WMS 提供 EMAP JPG 圖層；本函式將整體範圍切成多格，
    每格約 1200px，最後用 Pillow 無損對位拼接，再以高品質 JPEG 輸出。
    正式 HTTPS 網頁仍維持即時 NLSC WMTS，不使用這裡的快照。
    """
    west, south, east, north = [float(x) for x in bbox]
    lon_span = max(east - west, 1e-6)
    lat_span = max(north - south, 1e-6)
    center_lat = (south + north) / 2.0

    # 估算畫面比例，先把較長邊設到約 3600px。
    # 這比原本單張約 1800px 快照提高約 2 倍線性解析度。
    ground_ratio = max(
        lon_span * max(math.cos(math.radians(center_lat)), 0.2),
        1e-6,
    )
    target_width = max(2400, min(int(target_width), 4200))
    target_height = int(round(target_width * lat_span / ground_ratio))

    # 避免極端狹長範圍造成過大的 HTML。
    max_total_height = 4200
    if target_height > max_total_height:
        target_height = max_total_height
        target_width = int(round(target_height * ground_ratio / lat_span))
    target_width = max(2400, min(target_width, 4200))
    target_height = max(1800, min(target_height, max_total_height))

    # 每張 WMS 子圖控制在約 1200px，避免單張又被壓縮。
    tile_px = 1200
    cols = max(2, min(4, int(math.ceil(target_width / tile_px))))
    rows = max(2, min(4, int(math.ceil(target_height / tile_px))))

    tile_width = int(math.ceil(target_width / cols))
    tile_height = int(math.ceil(target_height / rows))

    # 重新讓每列／欄均勻對應到完整 bbox。
    tile_images = [[None for _ in range(cols)] for _ in range(rows)]

    ssl_fallback_reported = False

    def _download_one(r, c):
        nonlocal ssl_fallback_reported
        tile_west = west + lon_span * c / cols
        tile_east = west + lon_span * (c + 1) / cols
        tile_south = south + lat_span * r / rows
        tile_north = south + lat_span * (r + 1) / rows

        # 每個子區塊維持與整體 bbox 一致的畫面比例。
        tile_ground_ratio = max(
            (tile_east - tile_west)
            * max(math.cos(math.radians((tile_south + tile_north) / 2.0)), 0.2),
            1e-6,
        )
        tile_h = max(
            700,
            min(
                1400,
                int(round(tile_width * (tile_north - tile_south) / tile_ground_ratio)),
            ),
        )
        tile_w = tile_width

        params = {
            "SERVICE": "WMS",
            "VERSION": "1.1.1",
            "REQUEST": "GetMap",
            "LAYERS": "EMAP",
            "STYLES": "",
            "SRS": "EPSG:4326",
            "BBOX": (
                f"{tile_west:.10f},{tile_south:.10f},"
                f"{tile_east:.10f},{tile_north:.10f}"
            ),
            "WIDTH": str(tile_w),
            "HEIGHT": str(tile_h),
            "FORMAT": "image/jpeg",
            "TRANSPARENT": "FALSE",
            "DPI": "96",
            "MAP_RESOLUTION": "96",
            "FORMAT_OPTIONS": "dpi:96",
        }
        url = "https://wms.nlsc.gov.tw/wms?" + urllib.parse.urlencode(params)

        req = urllib.request.Request(
            url,
            headers={
                "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/140 Safari/537.36",
                "Accept": "image/jpeg,image/*;q=0.9,*/*;q=0.5",
            },
            method="GET",
        )

        try:
            with urllib.request.urlopen(req, timeout=45) as resp:
                data = resp.read()
                content_type = str(
                    resp.headers.get("Content-Type", "image/jpeg")
                ).lower()
        except ssl.SSLCertVerificationError as cert_exc:
            _insecure_ctx = ssl._create_unverified_context()
            with urllib.request.urlopen(
                req, timeout=45, context=_insecure_ctx
            ) as resp:
                data = resp.read()
                content_type = str(
                    resp.headers.get("Content-Type", "image/jpeg")
                ).lower()
            if not ssl_fallback_reported:
                print(
                    "[提示] NLSC 手機直開高解析底圖遇到 SSL 驗證問題；"
                    "本次快照改用相容性 SSL 模式。"
                )
                ssl_fallback_reported = True
        except urllib.error.URLError as url_exc:
            _reason = getattr(url_exc, "reason", url_exc)
            _reason_text = str(_reason)
            if "CERTIFICATE_VERIFY_FAILED" not in _reason_text:
                raise

            _insecure_ctx = ssl._create_unverified_context()
            with urllib.request.urlopen(
                req, timeout=45, context=_insecure_ctx
            ) as resp:
                data = resp.read()
                content_type = str(
                    resp.headers.get("Content-Type", "image/jpeg")
                ).lower()

            if not ssl_fallback_reported:
                print(
                    "[提示] NLSC 手機直開高解析底圖遇到 SSL 驗證問題；"
                    "本次快照改用相容性 SSL 模式。"
                )
                ssl_fallback_reported = True

        if not data or not data.startswith(b"\xff\xd8"):
            raise RuntimeError(
                "NLSC EMAP WMS 子圖沒有回傳有效 JPEG "
                f"(r={r}, c={c}, Content-Type={content_type}, bytes={len(data)})"
            )

        img = Image.open(BytesIO(data)).convert("RGB")
        return img, tile_w, tile_h

    # 順序下載，避免 Colab 同時建立太多 NLSC 請求。
    for r in range(rows):
        for c in range(cols):
            tile_images[r][c] = _download_one(r, c)

    stitched_width = cols * tile_width
    stitched_height = sum(tile_images[r][0][0].height for r in range(rows))
    canvas = Image.new("RGB", (stitched_width, stitched_height), "white")

    # WMS BBOX 使用 south->north，但影像座標是北在上。
    # 因此拼接時必須讓最北側的 row 放在 canvas 最上方，
    # 否則手機直開底圖會與路線的緯度方向上下顛倒。
    y = 0
    for display_r in range(rows):
        source_r = rows - 1 - display_r
        row_h = tile_images[source_r][0][0].height
        for c in range(cols):
            img = tile_images[source_r][c][0]
            if img.width != tile_width or img.height != row_h:
                img = img.resize((tile_width, row_h), Image.Resampling.LANCZOS)
            canvas.paste(img, (c * tile_width, y))
        y += row_h

    out = BytesIO()
    canvas.save(
        out,
        format="JPEG",
        quality=94,
        optimize=True,
        progressive=True,
    )
    data = out.getvalue()

    if not data or not data.startswith(b"\xff\xd8"):
        raise RuntimeError("NLSC 高解析 EMAP 拼接圖輸出失敗")

    return (
        data,
        "image/jpeg",
        (west, south, east, north),
        stitched_width,
        stitched_height,
    )


def _make_direct_phone_html_from_live_html(source_html, mobile_html, bbox):
    """由正式即時 HTML 另產一份「手機直接開 HTML」測試版。

    手機直開版採「即時 NLSC WMTS 優先＋內嵌高解析快照備援」：

    1. Cell 5 建立時先下載 NLSC EMAP 高解析 WMS 快照，直接 Base64 內嵌，
       確保 file:// 開啟時一定有可用底圖。
    2. HTML 開啟後，Leaflet 直接嘗試載入 NLSC EMAP WMTS 圖磚：
       https://wmts.nlsc.gov.tw/wmts/EMAP/default/GoogleMapsCompatible/{z}/{y}/{x}
    3. 只要至少一張 WMTS 圖磚成功載入，就移除快照，改用即時 NLSC 圖磚。
    4. 若 file:// 環境、瀏覽器或網路阻擋 WMTS，則保留內嵌快照。
    5. 不使用 fetch / XHR / API token；WMTS 由瀏覽器直接以圖片圖磚方式載入。

    正式 route_*_map.html 完全不改，仍使用原本即時 NLSC WMTS。
    """
    try:
        with open(source_html, "r", encoding="utf-8") as f:
            html = f.read()

        html = _make_html_standalone(html)

        image_bytes, content_type, bounds, width, height = _fetch_nlsc_emap_wms_snapshot(bbox)
        image_b64 = base64.b64encode(image_bytes).decode("ascii")
        west, south, east, north = bounds

        map_match = re.search(r'var\s+(map_[A-Za-z0-9_]+)\s*=\s*L\.map\(', html)
        if not map_match:
            raise RuntimeError("找不到 Leaflet map 變數，無法建立手機直開底圖。")
        map_var = map_match.group(1)

        payload = (
            "<script>\n"
            "(function(){\n"
            f"  var mapName = '{map_var}';\n"
            f"  var img = 'data:{content_type};base64,{image_b64}';\n"
            f"  var bounds = [[{south:.10f},{west:.10f}],[{north:.10f},{east:.10f}]];\n"
            "  var liveTileSuccess = false;\n"
            "  var liveTileLoadedCount = 0;\n"
            "  var liveTileErrors = 0;\n"
            "  var liveTileLayer = null;\n"
            "  var staticOverlay = null;\n"
            "  var liveTileUrl = 'https://wmts.nlsc.gov.tw/wmts/EMAP/default/GoogleMapsCompatible/{z}/{y}/{x}';\n"
            "\n"
            "  function addStaticBase(mapObj){\n"
            "    try {\n"
            "      if (!mapObj || !window.L || mapObj._cell5StaticBaseAdded) return true;\n"
            "      var pane = mapObj.getPane('cell5StaticBasePane');\n"
            "      if (!pane) {\n"
            "        mapObj.createPane('cell5StaticBasePane');\n"
            "        pane = mapObj.getPane('cell5StaticBasePane');\n"
            "      }\n"
            "      pane.style.zIndex = 205;\n"
            "      pane.style.pointerEvents = 'none';\n"
            "      staticOverlay = L.imageOverlay(img, bounds, {\n"
            "        pane:'cell5StaticBasePane',\n"
            "        interactive:false,\n"
            "        opacity:1.0,\n"
            "        zIndex:205\n"
            "      }).addTo(mapObj);\n"
            "      mapObj._cell5StaticBaseAdded = true;\n"
            "      return true;\n"
            "    } catch(e) {\n"
            "      console.warn('Cell5 手機直開內嵌底圖載入失敗', e);\n"
            "      return false;\n"
            "    }\n"
            "  }\n"
            "\n"
            "  function removeStaticBase(mapObj){\n"
            "    try {\n"
            "      if (staticOverlay && mapObj && mapObj.hasLayer(staticOverlay)) {\n"
            "        mapObj.removeLayer(staticOverlay);\n"
            "      }\n"
            "    } catch(e) {}\n"
            "    staticOverlay = null;\n"
            "  }\n"
            "\n"
            "  function addLiveNlscWmts(mapObj){\n"
            "    try {\n"
            "      if (!mapObj || !window.L || liveTileLayer) return true;\n"
            "      var pane = mapObj.getPane('cell5LiveWmtsPane');\n"
            "      if (!pane) {\n"
            "        mapObj.createPane('cell5LiveWmtsPane');\n"
            "        pane = mapObj.getPane('cell5LiveWmtsPane');\n"
            "      }\n"
            "      pane.style.zIndex = 215;\n"
            "      pane.style.pointerEvents = 'none';\n"
            "      liveTileLayer = L.tileLayer(liveTileUrl, {\n"
            "        minZoom: 0,\n"
            "        maxZoom: 19,\n"
            "        maxNativeZoom: 19,\n"
            "        tileSize: 256,\n"
            "        pane: 'cell5LiveWmtsPane',\n"
            "        opacity: 1.0,\n"
            "        updateWhenIdle: false,\n"
            "        keepBuffer: 2,\n"
            "        attribution: '© 內政部國土測繪中心'\n"
            "      });\n"
            "\n"
            "      liveTileLayer.on('tileload', function(){\n"
            "        liveTileLoadedCount++;\n"
            "        if (!liveTileSuccess && liveTileLoadedCount >= 2) {\n"
            "          liveTileSuccess = true;\n"
            "          try { removeStaticBase(mapObj); } catch(e) {}\n"
            "          console.info('Cell5：NLSC 即時 WMTS 已成功載入多張圖磚，改用即時圖資。');\n"
            "        }\n"
            "      });\n"
            "\n"
            "      liveTileLayer.on('tileerror', function(){\n"
            "        liveTileErrors++;\n"
            "        if (liveTileErrors >= 3 && liveTileLoadedCount < 2) {\n"
            "          console.warn('Cell5：NLSC 即時 WMTS 未形成可用圖磚，維持內嵌高解析快照。');\n"
            "        }\n"
            "      });\n"
            "\n"
            "      liveTileLayer.addTo(mapObj);\n"
            "      return true;\n"
            "    } catch(e) {\n"
            "      console.warn('Cell5：NLSC 即時 WMTS 建立失敗，維持內嵌快照。', e);\n"
            "      return false;\n"
            "    }\n"
            "  }\n"
            "\n"
            "  var tries=0;\n"
            "  function retry(){\n"
            "    var mapObj=window[mapName];\n"
            "    if(!mapObj){\n"
            "      if(++tries<30) setTimeout(retry,250);\n"
            "      return;\n"
            "    }\n"
            "    addStaticBase(mapObj);\n"
            "    addLiveNlscWmts(mapObj);\n"
            "    try { mapObj.invalidateSize({pan:false}); } catch(e) {}\n"
            "  }\n"
            "\n"
            "  if(document.readyState === 'loading') document.addEventListener('DOMContentLoaded', retry);\n"
            "  else retry();\n"
            "  setTimeout(retry,500);\n"
            "  setTimeout(retry,1500);\n"
            "  setTimeout(retry,3000);\n"
            "})();\n"
            "</script>\n"
        )

        html = html.replace("</body>", payload + "</body>", 1)

        with open(mobile_html, "w", encoding="utf-8") as f:
            f.write(html)

        return {
            "path": mobile_html,
            "snapshot": True,
            "width": width,
            "height": height,
            "live_wmts_test": True,
        }

    except Exception as exc:
        print(f"[警告] 建立手機直開 HTML 失敗：{exc}")
        raise



def _start_cell5_http_server(directory="/content", available_modes=None):
    """啟動 Cell 5 手機網頁服務。

    重要：
    1. 手機入口使用無副檔名 URL：/MOTOR、/CAR。
    2. HTTP 回應明確使用 text/html。
    3. 不送 Content-Disposition，避免 iPhone Edge 觸發下載。
    4. iframe 與一般瀏覽器連結都使用 /MOTOR / /CAR。
    """
    global _CELL5_HTTP_SERVER, _CELL5_HTTP_PORT, _CELL5_HTTP_MODES
    requested_modes = tuple(available_modes or [])
    if _CELL5_HTTP_SERVER is not None:
        if requested_modes == _CELL5_HTTP_MODES and _CELL5_HTTP_PORT is not None:
            return _CELL5_HTTP_PORT
        try:
            _CELL5_HTTP_SERVER.shutdown()
            _CELL5_HTTP_SERVER.server_close()
        except Exception:
            pass
        _CELL5_HTTP_SERVER = None
        _CELL5_HTTP_PORT = None
        _CELL5_HTTP_MODES = tuple()

    class _Cell5Handler(BaseHTTPRequestHandler):
        server_version = "Cell5Mobile/2.0-v61-Compatible"

        ROUTES = {}
        for _mode in (available_modes or []):
            ROUTES[f"/{_mode}"] = f"route_{_mode}_map.html"
            ROUTES[f"/{_mode.lower()}"] = f"route_{_mode}_map.html"

        def _send_html(self, body, status=200):
            if isinstance(body, str):
                body = body.encode("utf-8")
            self.send_response(status)
            # 關鍵：明確宣告 HTML，且完全不送 Content-Disposition。
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.send_header("Content-Length", str(len(body)))
            self.send_header("X-Content-Type-Options", "nosniff")
            self.send_header("Cache-Control", "no-store, no-cache, must-revalidate, max-age=0")
            self.send_header("Pragma", "no-cache")
            self.send_header("Referrer-Policy", "no-referrer")
            self.send_header("Permissions-Policy", "geolocation=(self)")
            self.end_headers()
            if self.command != "HEAD":
                self.wfile.write(body)

        def _load_map_html(self, filename):
            path = os.path.join(directory, filename)
            if not os.path.isfile(path):
                return None
            with open(path, "r", encoding="utf-8") as f:
                html = f.read()
            return _make_http_mobile_html(html)

        def _index_html(self):
            available = [m for m in (available_modes or []) if m in ("MOTOR", "CAR")]
            buttons = []
            if "MOTOR" in available:
                buttons.append('<a class="btn" href="/MOTOR">🏍️ MOTOR 機車地圖</a>')
            if "CAR" in available:
                buttons.append('<a class="btn" href="/CAR">🚗 CAR 汽車地圖</a>')
            buttons_html = "".join(buttons) or '<div>目前沒有可用的交通工具地圖。</div>'
            return f"""<!doctype html>
<html lang="zh-Hant">
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width,initial-scale=1,viewport-fit=cover">
<title>農地順順走地圖</title>
<style>
body{{font-family:-apple-system,BlinkMacSystemFont,"Microsoft JhengHei",Arial,sans-serif;margin:0;padding:24px;background:#f5f5f5;color:#222}}
main{{max-width:520px;margin:0 auto;background:#fff;border-radius:14px;padding:20px;box-shadow:0 2px 12px rgba(0,0,0,.12)}}
h1{{font-size:22px;margin:0 0 16px}}
.btn{{display:block;text-decoration:none;text-align:center;padding:14px;margin:10px 0;border:1px solid #999;border-radius:10px;color:#111;background:#fafafa;font-weight:700;font-size:17px}}
small{{color:#666;line-height:1.5}}
</style>
</head>
<body><main>
<h1>農地順順走地圖</h1>
{buttons_html}
<small>請使用 Microsoft Edge。這裡是直接渲染 HTML 的網頁入口，不是檔案下載連結。</small>
</main></body></html>"""

        def do_HEAD(self):
            self._handle_request()

        def do_GET(self):
            self._handle_request()

        def _handle_request(self):
            parsed = urllib.parse.urlparse(self.path)
            path = urllib.parse.unquote(parsed.path).rstrip("/") or "/"
            query = urllib.parse.parse_qs(parsed.query)

            # 公開 Tunnel 健康檢查：只有這裡可從公開網址回到 Colab 時才顯示手機網址。
            if path == "/__cell5_health":
                body = b"CELL5_OK"
                self.send_response(200)
                self.send_header("Content-Type", "text/plain; charset=utf-8")
                self.send_header("Content-Length", str(len(body)))
                self.send_header("Cache-Control", "no-store, no-cache, must-revalidate, max-age=0")
                self.send_header("X-Content-Type-Options", "nosniff")
                self.end_headers()
                if self.command != "HEAD":
                    self.wfile.write(body)
                return

            # 舊 .html 入口：改導向 query viewer，避免部分 iOS Edge 將檔案型 URL 當成下載。
            if path in ("/route_MOTOR_map.html", "/route_CAR_map.html"):
                target_mode = "MOTOR" if "MOTOR" in path else "CAR"
                if target_mode in (available_modes or []):
                    self.send_response(302)
                    self.send_header("Location", "/?mode=" + target_mode)
                    self.send_header("Cache-Control", "no-store")
                    self.end_headers()
                else:
                    self._send_html("<h1>此交通工具地圖尚未產生</h1>", 404)
                return

            mode = str((query.get("mode") or [""])[0]).upper()
            if path == "/" and mode in ("MOTOR", "CAR"):
                filename = self.ROUTES["/" + mode]
                html = self._load_map_html(filename)
                if html is None:
                    self._send_html("<h1>地圖尚未產生</h1>", 404)
                else:
                    self._send_html(html)
                return

            if path == "/":
                self._send_html(self._index_html())
                return

            filename = self.ROUTES.get(path)
            if filename:
                # 正式入口仍保留，但同步提供 query viewer 作為 iPhone Edge 首選。
                html = self._load_map_html(filename)
                if html is None:
                    self._send_html("<h1>地圖尚未產生</h1>", 404)
                else:
                    self._send_html(html)
                return

            self._send_html("<h1>404</h1>", 404)

        def log_message(self, format, *args):
            pass

    last_error = None
    for _port in range(8765, 8790):
        try:
            server = ThreadingHTTPServer(("0.0.0.0", _port), _Cell5Handler)
            threading.Thread(target=server.serve_forever, daemon=True).start()
            _CELL5_HTTP_SERVER = server
            _CELL5_HTTP_PORT = _port
            _CELL5_HTTP_MODES = requested_modes
            return _port
        except OSError as e:
            last_error = e
    raise OSError(f"無法啟動 Cell 5 HTTP 服務：{last_error}")


def _ensure_cloudflared_binary():
    """確保 Colab 內有可執行的 cloudflared（Linux amd64）。"""
    global _CELL5_CLOUDFLARED
    if os.path.isfile(_CELL5_CLOUDFLARED) and os.access(_CELL5_CLOUDFLARED, os.X_OK):
        return _CELL5_CLOUDFLARED

    url = "https://github.com/cloudflare/cloudflared/releases/latest/download/cloudflared-linux-amd64"
    tmp_path = _CELL5_CLOUDFLARED + ".download"
    print("[手機網頁] 正在準備 cloudflared…")
    urllib.request.urlretrieve(url, tmp_path)
    os.replace(tmp_path, _CELL5_CLOUDFLARED)
    os.chmod(_CELL5_CLOUDFLARED, 0o755)
    return _CELL5_CLOUDFLARED


def _stop_existing_cell5_tunnel():
    global _CELL5_TUNNEL_PROCESS
    proc = _CELL5_TUNNEL_PROCESS
    if proc is not None:
        try:
            if proc.poll() is None:
                proc.terminate()
                proc.wait(timeout=3)
        except Exception:
            try:
                proc.kill()
            except Exception:
                pass
    _CELL5_TUNNEL_PROCESS = None


def _start_cell5_cloudflare_tunnel(port):
    """建立 TryCloudflare 公開 HTTPS 網址，供手機 Edge 直接開啟。

    本版刻意沿用已實際在 iPhone Microsoft Edge 成功的 v18 Tunnel 流程：
    不增加 DNS over HTTPS、curl --resolve 或公開健康檢查。
    取得 trycloudflare.com 網址後直接保留 Tunnel，避免 Colab 端額外
    的 DNS / 出口檢查反而把可用 Tunnel 判定成失敗並終止。
    """
    global _CELL5_TUNNEL_PROCESS, _CELL5_MOBILE_BASE_URL

    _stop_existing_cell5_tunnel()
    cloudflared = _ensure_cloudflared_binary()
    log_path = _CELL5_TUNNEL_LOG

    try:
        if os.path.exists(log_path):
            os.remove(log_path)
    except Exception:
        pass

    cmd = [
        cloudflared,
        "tunnel",
        "--no-autoupdate",
        "--protocol",
        "http2",
        "--url",
        f"http://127.0.0.1:{int(port)}",
        "--logfile",
        log_path,
    ]

    _CELL5_TUNNEL_PROCESS = subprocess.Popen(
        cmd,
        stdin=subprocess.DEVNULL,
        stdout=subprocess.DEVNULL,
        stderr=subprocess.DEVNULL,
        start_new_session=True,
    )

    url_pattern = re.compile(r"https://[a-z0-9-]+\.trycloudflare\.com", re.I)
    deadline = time.time() + 30
    found_url = None

    while time.time() < deadline:
        if _CELL5_TUNNEL_PROCESS.poll() is not None:
            break
        try:
            with open(log_path, "r", encoding="utf-8", errors="ignore") as f:
                log_text = f.read()
            matches = url_pattern.findall(log_text)
            if matches:
                found_url = matches[-1].rstrip("/")
                break
        except FileNotFoundError:
            pass
        except Exception:
            pass
        time.sleep(0.25)

    if not found_url:
        _stop_existing_cell5_tunnel()
        tail = ""
        try:
            with open(log_path, "r", encoding="utf-8", errors="ignore") as f:
                tail = f.read()[-4000:]
        except Exception:
            pass
        raise RuntimeError(
            "Cloudflare Quick Tunnel 未能產生手機網址。"
            + (f"\ncloudflared log:\n{tail}" if tail else "")
        )

    # 重要：不要再做 Colab 端的公開 DNS / HTTPS 健康檢查。
    # v18 已證明這條 Tunnel 模式可由 iPhone Edge 實際開啟；
    # v18 的健康檢查反而可能因 Colab DNS 解析不到 trycloudflare.com
    # 而自行終止原本已成功建立的 Tunnel。
    _CELL5_MOBILE_BASE_URL = found_url
    return found_url



def _available_vehicle_modes(map_outputs):
    """依目前實際產生的地圖輸出，回傳可提供 HTTPS 的交通工具模式。"""
    modes = []
    for mode in ("MOTOR", "CAR"):
        if mode in (map_outputs or {}) and os.path.isfile(str(map_outputs[mode])):
            modes.append(mode)
    return modes

def _show_cell5_mobile_links(map_outputs):

    print("此網址會在目前 Colab 工作階段結束後失效。")

    try:
        public_base = _start_cell5_cloudflare_tunnel(port)
    except Exception as e:

    print()

    for mode in modes:

        print(public_base + f"/{mode}")
        print(f"備援（query）{mode}：")

    print()
    print("請把對應交通工具網址貼到 iPhone Microsoft Edge，會直接開成網頁。")
    print("不要下載 route_*_map.html。")

    _display_clickable_mobile_links(public_base, modes)

    try:
        from google.colab.output import eval_js
        base_url = str(eval_js(f"google.colab.kernel.proxyPort({port})")).rstrip("/")
        print()
        print("桌面 Colab proxy 測試入口：")
        for mode in modes:
            print(f"{mode}：{base_url}/{mode}")
    except Exception:
        pass


def build_vehicle_map(payload, cadastral_gdf):
    vehicle_mode = payload["VEHICLE_MODE"]
    vehicle_label = payload["VEHICLE_LABEL"]
    work_df = payload["work_df"]
    coords = payload["coords"]
    snap_distances = payload["snap_distances"]
    start_name = payload["start_name"]
    start_lat = float(payload["start_lat"])
    start_lon = float(payload["start_lon"])
    route_order_df = payload["route_order_df"]
    # 顯示名稱／案件資料一律回到 Cell 4 work_df 取得 canonical 版本。
    _work_lookup_idx, _work_lookup_case, _work_lookup_orig = _build_work_lookup(work_df)
    actual_batches = payload["actual_batches"]
    return_infos = payload["return_infos"]
    actual_connectors = payload["actual_connectors"]
    all_in_one_order = payload["all_in_one_order"]
    single_day_preferred = str(payload.get("single_day_preferred", "daily") or "daily")
    single_day_is_active = (len(actual_batches) == 1)
    all_in_one_geometry = payload["all_in_one_geometry"]
    all_in_one_return_geometry = payload["all_in_one_return_geometry"]
    all_in_one_return_pair = payload["all_in_one_return_pair"]
    all_in_one_actual_distance = float(payload["all_in_one_actual_distance"])
    all_in_one_actual_drive_min = float(payload["all_in_one_actual_drive_min"])

    center_lat = float(coords[:, 1].mean())
    center_lon = float(coords[:, 0].mean())

    m = folium.Map(
        location=[center_lat, center_lon],
        zoom_start=10,
        control_scale=True,
        tiles=None,
    )

    # v16：縮小搜尋／路線資訊、圖面選單與勘查順序點位框。
    _layer_items = []

    # ========================================================
    # NLSC 底圖與地籍向量圖層
    # ========================================================
    NLSC_BASE = "https://wmts.nlsc.gov.tw/wmts"

    def add_nlsc_base(name, layer, show=False, max_zoom=19):
        _layer = folium.TileLayer(
            tiles=(
                f"{NLSC_BASE}/{layer}/default/"
                "GoogleMapsCompatible/{z}/{y}/{x}"
            ),
            attr="國土測繪中心 NLSC",
            name=name,
            overlay=False,
            control=False,
            show=show,
            max_zoom=max_zoom,
        ).add_to(m)
        _layer_items.append({
            "type": "base",
            "label": name,
            "var": _layer.get_name(),
            "show": bool(show),
        })
        return _layer

    add_nlsc_base("EMAP｜臺灣通用電子地圖", "EMAP", show=True)
    add_nlsc_base("EMAP6｜不含等高線", "EMAP6", show=False)
    add_nlsc_base("PHOTO2｜正射影像", "PHOTO2", show=False)


    _landsect_layer = folium.TileLayer(
        tiles=(
            f"{NLSC_BASE}/LANDSECT/default/"
            "GoogleMapsCompatible/{z}/{y}/{x}"
        ),
        attr="國土測繪中心 NLSC",
        name="LANDSECT｜地段外圍圖",
        overlay=True,
        control=False,
        show=False,
        max_zoom=18,
        opacity=0.82,
    ).add_to(m)
    _layer_items.append({
        "type": "overlay",
        "label": "LANDSECT｜地段外圍圖",
        "var": _landsect_layer.get_name(),
        "show": False,
        "group": "other",
    })


    # ========================================================
    # Cell 3 地籍向量 Polygon
    # ========================================================
    cad_map_gdf = _prepare_cadastral_map_gdf(
        cadastral_gdf,
        route_order_df,
    )
    start_display_name = _resolve_start_display_name(
        route_order_df,
        start_name,
        work_df=work_df,
        cadastral_map_gdf=cad_map_gdf,
    )
    cad_geojson = _cadastral_geojson(cad_map_gdf)

    day_colors = [
        "#d62728", "#1f77b4", "#2ca02c", "#9467bd", "#ff7f0e",
        "#8c564b", "#17becf", "#006d2c", "#3182bd", "#e377c2", "#7f7f7f",
    ]
    day_color_map = {
        day: day_colors[(day - 1) % len(day_colors)]
        for day in range(1, 100)
    }

    def cad_style(feature):
        day = feature.get("properties", {}).get("工作日")
        try:
            day = int(float(day))
        except Exception:
            day = 0
        color = day_color_map.get(day, "#666666")
        return {
            "color": color,
            "weight": 2,
            "opacity": 0.90,
            "fillColor": color,
            "fillOpacity": 0.10,
        }

    def cad_highlight(feature):
        return {
            "weight": 4,
            "color": "#111111",
            "fillOpacity": 0.22,
        }

    cad_fields = [
        c for c in [
            "案件ID", "縣市", "鄉鎮市區", "地段", "地號", "名稱",
            "面積_m2", "解析度分類", "定位點位於宗地", "中心偏移_m",
            "工作日", "日內序號", "總路線序號",
        ] if c in cad_map_gdf.columns
    ]
    cad_alias_map = {
        "案件ID": "案件ID",
        "縣市": "縣市",
        "鄉鎮市區": "鄉鎮市區",
        "地段": "地段",
        "地號": "地號",
        "名稱": "名稱",
        "面積_m2": "面積(m²)",
        "解析度分類": "向量解析度",
        "定位點位於宗地": "定位點位於宗地",
        "中心偏移_m": "定位點－中心距離(m)",
        "工作日": "工作日",
        "日內序號": "日內序號",
        "總路線序號": "勘查編號",
    }
    cad_aliases = [cad_alias_map[c] for c in cad_fields]

    # --------------------------------------------------------
    # 地籍 Polygon 圖層
    #
    # 重要：地址案件可能完全沒有 Polygon。這種情況下不能建立
    # folium.GeoJson(empty FeatureCollection, tooltip=...)，因為 Folium
    # 會從空資料找欄位並在 render() 時觸發：
    #   AssertionError: The field 案件ID is not available in the data.
    #
    # 因此：
    #   有 Polygon → 正常建立 GeoJson + Tooltip。
    #   無 Polygon → 只建立一個隱藏的空 FeatureGroup 供 UI 聯動使用，
    #               不建立 GeoJson、不建立 Tooltip、不加入圖層清單。
    # --------------------------------------------------------
    cad_layer = folium.FeatureGroup(
        name="NLSC 地籍向量（Polygon）",
        show=False,
        overlay=True,
        control=True,
    )

    if CADASTRAL_VECTOR_AVAILABLE and not cad_map_gdf.empty:
        cad_geojson_layer = folium.GeoJson(
            cad_geojson,
            name="NLSC 地籍向量",
            style_function=cad_style,
            smooth_factor=0.5,
            tooltip=(
                folium.GeoJsonTooltip(
                    fields=cad_fields,
                    aliases=cad_aliases,
                    localize=True,
                    sticky=False,
                    labels=True,
                    style=(
                        "background-color:white; color:#111; "
                        "font-family:Microsoft JhengHei,Arial; font-size:12px;"
                    ),
                )
                if cad_fields else None
            ),
            # 地籍向量只保留滑鼠移入 Tooltip，不在點擊時開啟 Popup。
            # 避免與「勘查順序」Marker 疊在同一位置時，同一次點擊跳出兩個提示框。
            popup=None,
            zoom_on_click=False,
        )
        cad_geojson_layer.add_to(cad_layer)
        cad_layer.options.update({"show": True})

        cad_layer.add_to(m)
        _layer_items.append({
            "type": "overlay",
            "label": "NLSC 地籍向量（Polygon）",
            "var": cad_layer.get_name(),
            "show": True,
            "group": "other",
        })
        cad_ui_layer_var = cad_layer.get_name()
    else:
        # 沒有 Polygon 時，FeatureGroup 未加入地圖，因此 Folium 不會輸出
        # 該 FeatureGroup 的 JS 變數。Cell 5 JS 必須使用 null，不能引用
        # 不存在的 feature_group_xxx，否則會讓後續 UI（我的位置、面板）
        # 整段初始化中止。
        cad_ui_layer_var = "null"

    # ========================================================
    # 勘查編號：預設顯示
    # ========================================================
    point_number_layer = folium.FeatureGroup(
        name="勘查編號",
        show=False,
        overlay=True,
        control=True,
    )

    route_marker_refs = []
    search_records = []
    marker_links = {}
    inspection_records = {}
    inspection_meta = {}
    route_arrow_bindings = []

    _inspection_no = 0
    for _, _raw_row in route_order_df.iterrows():
        row = _canonical_route_row(
            _raw_row,
            _work_lookup_idx,
            _work_lookup_case,
            _work_lookup_orig,
        )
        if str(row.get("角色", "")).strip() == "固定起點":
            continue

        _inspection_no += 1
        route_no = _inspection_no
        day_no = int(row["工作日"])
        day_seq = int(row["日內序號"])
        name_v = escape(str(row["名稱"]))
        sect_v = escape(str(row["地段"]))
        land_v = escape(str(row["地號"]))
        lat = float(row["緯度"])
        lon = float(row["經度"])

        _display_name = escape(_standardized_cadastral_name(row))
        popup = (
            f"<div class=\"cell5-inspection-name\">{_display_name}</div>"
            + _popup_area_html(row)
            + google_links_html(lat, lon, vehicle_mode)
        )

        number_html = f'''
        <div style="width:28px;height:28px;line-height:28px;
            border:2px solid #333;border-radius:50%;background:rgba(255,255,255,0.94);
            color:#111;font-size:13px;font-weight:700;text-align:center;
            box-shadow:0 1px 4px rgba(0,0,0,0.35);
            font-family:Arial,Microsoft JhengHei,sans-serif;">{route_no}</div>'''

        marker = folium.Marker(
            location=[lat, lon],
            z_index_offset=int(10000000 - route_no * 10000),
            icon=folium.DivIcon(
                html=number_html,
                icon_size=(23, 23),
                icon_anchor=(11, 11),
                class_name="route-point-number",
            ),
            popup=folium.Popup(popup, max_width=500),
            interactive=True,
            tooltip=(
                f"勘查編號：{route_no}｜"
                f"DAY{day_no:02d}－{day_seq}｜{_standardized_cadastral_name(row)}"
            ),
        ).add_to(point_number_layer)
        route_marker_refs.append((marker, route_no))

        point_index_v = _safe_num(row.get("點索引", -1), -1)
        case_id_v = str(row.get("案件ID", ""))

        # 「點索引」作為主要聯動鍵，避免案件ID重複造成錯誤或定位到錯誤 Polygon。
        marker_links[marker.get_name()] = {
            "point_index": point_index_v,
            "case_id": case_id_v,
        }
        inspection_records[marker.get_name()] = popup

        search_records.append({
            "keys": [
                str(route_no),
                str(day_no),
                str(day_seq),
                str(row.get("名稱", "")),
                str(row.get("地段", "")),
                str(row.get("地號", "")),
                str(row.get("案件ID", "")),
                str(row.get("點索引", "")),
            ],
            "point_index": point_index_v,
            "case_id": case_id_v,
            "label": f"#{route_no}｜{_standardized_cadastral_name(row)}",
            "marker_var": marker.get_name(),
            "marker": marker,
            "lat": lat,
            "lon": lon,
        })


    # 「勘查編號」預設關閉，但仍必須 add_to(m) 讓 Folium 將 FeatureGroup
    # 與其內的 Marker 宣告寫入 HTML。這樣自訂「地圖顯示內容」勾選後，
    # JavaScript 才能透過 map.addLayer()/removeLayer() 真正控制整組點位。
    # show=False：仍維持初始不顯示。
    point_number_layer.add_to(m)
    _layer_items.append({
        "type": "overlay",
        "label": "勘查編號",
        "var": point_number_layer.get_name(),
        "show": False,
        "group": "number",
    })

    # ========================================================
    # 共同固定起點：所有 DAY 與一次完成共用，同一張地圖只顯示一次。
    # ========================================================
    start_layer = folium.FeatureGroup(
        name="固定起點",
        show=True,
        overlay=True,
        control=True,
    )
    _shared_start_html = '''
    <div style="width:28px;height:28px;line-height:28px;
        border:2px solid #163a72;border-radius:50%;background:rgba(235,242,255,0.96);
        color:#163a72;font-size:13px;font-weight:700;text-align:center;
        box-shadow:0 1px 4px rgba(0,0,0,0.35);">0</div>'''
    _shared_start_marker = folium.Marker(
        location=[start_lat, start_lon],
        tooltip=f"固定起點：0｜{start_display_name}",
        popup=None,
        icon=folium.DivIcon(
            html=_shared_start_html,
            icon_size=(28, 28),
            icon_anchor=(14, 14),
            class_name="cell5-shared-start-marker",
        ),
        z_index_offset=30000000,
    ).add_to(start_layer)
    # 重要：Cell 4 的第一筆有效資料就是固定起點，而且 route_order_df 的
    # 固定起點列本身沒有「案件ID」欄位。為了維持跨 Cell 案件ID完整性，
    # 必須從 Cell 4 canonical work_df 取得固定起點的案件ID，掛到共用起點
    # Marker 的 marker_links metadata。這不改變地圖顯示或路線邏輯。
    _fixed_start_case_id = ""
    try:
        if isinstance(work_df, pd.DataFrame) and not work_df.empty:
            _fixed_start_case_id = _clean_display_text(
                work_df.iloc[0].get("案件ID", "")
            )
    except Exception:
        _fixed_start_case_id = ""
    marker_links[_shared_start_marker.get_name()] = {
        "point_index": -1,
        "case_id": _fixed_start_case_id,
    }
    shared_start_marker_var = _shared_start_marker.get_name()

    # 全域「勘查編號」也加入資訊面板的上一點／下一點。
    for _gi, (_gmarker, _groute_no) in enumerate(route_marker_refs):
        _gvar = _gmarker.get_name()
        # v33.18：全域第 1 點沒有上一點，因此不可點選。
        _gprev = "" if _gi == 0 else route_marker_refs[_gi - 1][0].get_name()
        _gnext = route_marker_refs[_gi + 1][0].get_name() if _gi + 1 < len(route_marker_refs) else ""
        _gprev_btn = (
            f"<button type='button' class='cell5-nav-button' data-cell5-nav-marker='{escape(_gprev, quote=True)}'>上一點</button>"
            if _gprev else '<button type="button" class="cell5-nav-button" disabled>上一點</button>'
        )
        _gnext_btn = (
            f"<button type='button' class='cell5-nav-button' data-cell5-nav-marker='{escape(_gnext, quote=True)}'>下一點</button>"
            if _gnext else '<button type="button" disabled>下一點</button>'
        )
        _gvar = _gmarker.get_name()
        inspection_records[_gvar] = (
            inspection_records.get(_gvar, "")
            + f'<div class="cell5-inspection-nav">{_gprev_btn}{_gnext_btn}</div>'
            + '<div class="cell5-inspection-nav-hint">依全域勘查編號導航</div>'
        )
        _glink = marker_links.get(_gvar, {}) or {}
        _gpoint_index = int(_safe_num(_glink.get("point_index", -1), -1))
        _gmeta_row = _work_lookup_idx.get(_gpoint_index)
        if _gmeta_row is None:
            # 正常情況 work_df 一定能以「點索引」找到；若缺失，至少從 route_order_df 當前列補資料。
            _gmeta_row = _canonical_route_row(
                route_order_df.iloc[min(_gi, max(0, len(route_order_df) - 1))],
                _work_lookup_idx,
                _work_lookup_case,
                _work_lookup_orig,
            )
        inspection_meta[_gvar] = {
            "point_index": int(_safe_num(_glink.get("point_index", -1), -1)),
            "label": f"勘查編號 #{_groute_no}｜全域順序",
            "display_name": _standardized_cadastral_name(_gmeta_row),
            "lat": float(_gmeta_row.get("緯度", 0.0)),
            "lon": float(_gmeta_row.get("經度", 0.0)),
            "return_start_url": google_maps_direction_url(start_lat, start_lon, vehicle_mode),
        }

    # --------------------------------------------------------
    # v33.12：把「勘查編號」的上一點／下一點同步寫回「真正的
    # Leaflet Popup」。
    #
    # 前段建立 Marker 時，Popup 已經先建立完成；後面才補入
    # inspection_records 的全域上一點／下一點，因此只改資料字典
    # 不會自動改到已生成的 Folium Popup。
    # 這裡直接更新每個 Marker 原本 Popup 內的 Html.data，
    # 讓使用者點開「勘查編號」時，Popup 本身就會看到上一點／下一點。
    # --------------------------------------------------------
    for _gmarker, _groute_no in route_marker_refs:
        _gvar = _gmarker.get_name()
        _gpopup_html = inspection_records.get(_gvar, "")
        if not _gpopup_html:
            continue
        try:
            for _child in _gmarker._children.values():
                if _child.__class__.__name__ != "Popup":
                    continue
                if hasattr(_child, "html") and hasattr(_child.html, "_children"):
                    _html_children = list(_child.html._children.values())
                    if _html_children and hasattr(_html_children[0], "data"):
                        _html_children[0].data = _gpopup_html
                break
        except Exception as _popup_sync_err:
            print(f"[警告] 勘查編號 Popup 導航同步失敗（#{_groute_no}）：{_popup_sync_err}")

    start_layer.add_to(m)
    _layer_items.append({
        "type": "overlay",
        "label": "固定起點",
        "var": start_layer.get_name(),
        "show": True,
        "group": "other",
    })

    # ========================================================
    # 每日路線與每日勘查順序
    # ========================================================
    colors = [
        "red", "blue", "green", "purple", "orange", "darkred",
